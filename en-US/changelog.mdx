<script>
(function () {
  var root = document.getElementById("changelog-root");
  var REPO = "molesignal/molesignal";
  var PER_PAGE = 20;

  function esc(s) {
    return String(s)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;");
  }

  /* Lightweight GitHub-flavoured markdown → HTML. Strips HTML tags first. */
  function md2html(md) {
    var raw = String(md || "")
      .replace(/<[^>]*>/g, "");           // strip any raw HTML

    /* Split into blocks (double-newline). */
    var blocks = raw.split(/\n{2,}/);
    var out = [];
    var inTable = false;
    var inCode = false;

    for (var i = 0; i < blocks.length; i++) {
      var b = blocks[i].trim();
      if (!b) continue;

      /* Fenced code blocks */
      if (/^```/.test(b)) {
        if (inCode) { out.push("</code></pre>"); inCode = false; continue; }
        inCode = true;
        var lang = b.replace(/^```\s*/, "") || "";
        out.push('<pre><code' + (lang ? ' class="language-' + esc(lang) + '"' : "") + ">");
        continue;
      }
      if (inCode) { out.push(esc(b) + "\n"); continue; }

      /* Table */
      if (/\|.*\|/.test(b) && b.indexOf("\n") === -1) {
        var rows = [b];
        while (i + 1 < blocks.length && /\|.*\|/.test(blocks[i + 1].trim())) {
          rows.push(blocks[++i].trim());
        }
        out.push(renderTable(rows));
        continue;
      }

      /* Headings */
      if (/^####\s/.test(b))  { out.push("<h4>" + inline(b.replace(/^####\s*/, "")) + "</h4>"); continue; }
      if (/^###\s/.test(b))   { out.push("<h3>" + inline(b.replace(/^###\s*/, "")) + "</h3>"); continue; }
      if (/^##\s/.test(b))    { out.push("<h2>" + inline(b.replace(/^##\s*/, "")) + "</h2>"); continue; }

      /* Unordered list */
      if (/^-\s/.test(b) && b.indexOf("\n") === -1) {
        var items = [b];
        while (i + 1 < blocks.length && /^-\s/.test(blocks[i + 1].trim())) {
          items.push(blocks[++i].trim());
        }
        out.push("<ul>" + items.map(function(it) {
          return "<li>" + inline(it.replace(/^-\s*/, "")) + "</li>";
        }).join("") + "</ul>");
        continue;
      }

      /* Horizontal rule */
      if (/^---$/.test(b)) { out.push("<hr>"); continue; }

      /* Blockquote */
      if (/^>\s/.test(b)) {
        out.push("<blockquote>" + inline(b.replace(/^>\s*/, "")) + "</blockquote>");
        continue;
      }

      /* Paragraph */
      out.push("<p>" + inline(b) + "</p>");
    }

    if (inCode) out.push("</code></pre>");
    return out.join("\n");
  }

  function inline(s) {
    return s
      .replace(/`([^`]+)`/g, "<code>$1</code>")
      .replace(/!\[([^\]]*)\]\(([^)]+)\)/g, '<img alt="$1" src="$2" loading="lazy">')
      .replace(/\[([^\]]+)\]\(([^)]+)\)/g, '<a href="$2" target="_blank" rel="noopener">$1</a>')
      .replace(/\*\*([^*]+)\*\*/g, "<strong>$1</strong>")
      .replace(/\*([^*]+)\*/g, "<em>$1</em>")
      .replace(/__([^_]+)__/g, "<strong>$1</strong>")
      .replace(/_([^_]+)_/g, "<em>$1</em>")
      .replace(/~~([^~]+)~~/g, "<del>$1</del>")
      .replace(/\n/g, "<br>");
  }

  function renderTable(rows) {
    if (rows.length < 2) return "";
    /* Skip separator row like |---|---| */
    var sepIdx = rows[1] && /^\|[\s\-:|]+\|$/.test(rows[1]) ? 1 : -1;
    var header = sepIdx >= 0 ? rows[0] : null;
    var body = sepIdx >= 0 ? rows.slice(2) : rows;
    var h = "";
    if (header) {
      h = "<thead><tr>" + header.split("|").filter(Boolean).map(function(c) {
        return "<th>" + inline(c.trim()) + "</th>";
      }).join("") + "</tr></thead>";
    }
    var b = "<tbody>" + body.map(function(row) {
      return "<tr>" + row.split("|").filter(Boolean).map(function(c) {
        return "<td>" + inline(c.trim()) + "</td>";
      }).join("") + "</tr>";
    }).join("") + "</tbody>";
    return "<table>" + h + b + "</table>";
  }

  function fmtDate(iso) {
    if (!iso) return "";
    return iso.slice(0, 10);
  }

  function render(releases) {
    if (!releases.length) {
      return '<p>No releases yet. <a href="https://github.com/' + REPO + '/releases" target="_blank" rel="noopener">View on GitHub</a>.</p>';
    }
    return releases.map(function (r) {
      var date = fmtDate(r.published_at);
      var tag = esc(r.tag_name || r.name || "");
      var prerelease = r.prerelease
        ? ' <mark class="changelog-prerelease">pre-release</mark>'
        : "";
      var body = md2html(r.body) || "<p><em>No release notes.</em></p>";
      var url = r.html_url || "https://github.com/" + REPO + "/releases/tag/" + encodeURIComponent(r.tag_name);

      return (
        '<article class="changelog-release">' +
        '  <header class="changelog-header">' +
        '    <h2 class="changelog-title">' +
        '      <a href="' + url + '" target="_blank" rel="noopener">' + tag + '</a>' +
        '    </h2>' +
        '    <div class="changelog-meta">' +
        '      <time datetime="' + esc(r.published_at || "") + '">' + esc(date) + "</time>" +
        prerelease +
        "    </div>" +
        "  </header>" +
        '  <div class="changelog-body">' + body + "</div>" +
        "</article>"
      );
    }).join("\n");
  }

  /* --- main --- */
  root.innerHTML =
    '<p style="color:var(--mint-text-secondary, #64748b);padding:1rem 0">Loading releases…</p>';

  fetch("https://api.github.com/repos/" + REPO + "/releases?per_page=" + PER_PAGE)
    .then(function (res) {
      if (!res.ok) throw new Error("GitHub API returned " + res.status);
      return res.json();
    })
    .then(function (releases) {
      root.innerHTML = render(releases);
    })
    .catch(function (err) {
      root.innerHTML =
        '<div class="changelog-error" style="padding:1rem;border-left:3px solid #ef4444;background:rgba(239,68,68,0.06);border-radius:0 6px 6px 0">' +
        "<p><strong>Unable to load releases.</strong></p>" +
        '<p style="font-size:0.875rem;color:var(--mint-text-secondary,#64748b)">' +
        '<a href="https://github.com/' + REPO + '/releases" target="_blank" rel="noopener">View on GitHub →</a>' +
        "</p></div>";
      console.error("changelog:", err);
    });
})();
</script>
